const (
	smallPart = 1024
	swsz      = 384
	unroll    = 16
)

func b2i(b bool) int {
	var i int
	if b {
		i = 1
	}
	return i
}

// pivotJitter picks the 0..7 offset of the pivot samples. The C version uses
// a global PRNG, which would be a data race in Go; a hash of the partition
// bounds is deterministic (like the stdlib pdqsort) and the heapsort fallback
// still bounds the worst case.
func pivotJitter(left, right int) int {
	h := uint64(left)*0x9E3779B97F4A7C15 ^ uint64(right)
	h ^= h >> 29
	return int(h * 0xBF58476D1CE4E5B9 >> 61)
}

// depthLimit is the number of lopsided splits sortr tolerates before
// falling back to heapsort.
func depthLimit(n int) int {
	return bits.Len(uint(n))
}

// runLimit is the number of monotone runs (as counted by scanOrder) up to
// which pdqsort's pattern detection beats blqsort; measured on amd64.
func runLimit(n int) int {
	if n <= 1024 {
		return n / 16
	}
	return 16
}

func s2Func[T any](cmp func(a, b T) int, x, y T) (T, T) {
	if cmp(y, x) < 0 {
		return y, x
	}
	return x, y
}

func heapSortFunc[T any](a []T, cmp func(a, b T) int) {
	n := len(a)
	if n < 2 {
		return
	}
	for i := n / 2; ; {
		var k T
		if i > 0 {
			i--
			k = a[i]
		} else {
			n--
			if n == 0 {
				return
			}
			k = a[n]
			a[n] = a[0]
		}
		j := i
		for j*2+1 < n {
			child := j*2 + 1
			if child+1 < n && cmp(a[child], a[child+1]) < 0 {
				child++
			}
			if cmp(k, a[child]) >= 0 {
				break
			}
			a[j] = a[child]
			j = child
		}
		a[j] = k
	}
}

func partitionSmallFunc[T any](a []T, left, right int, buf []T, cmp func(a, b T) int) int {
	outerleft := left
	pivp := left + (right-left)/2
	piv := a[pivp]

	// Median of 7 samples spread evenly over the range. The C version samples
	// both ends and the middle (and then skips the end samples), which picks
	// extreme pivots on inputs with positional structure.
	step := (right - left) / 8
	p0, p1, p2 := pivp-3*step, pivp-2*step, pivp-step
	p4, p5, p6 := pivp+step, pivp+2*step, pivp+3*step
	a[p0], a[p1], a[p2], piv, a[p4], a[p5], a[p6] = med7Func(cmp, a[p0], a[p1], a[p2], piv, a[p4], a[p5], a[p6])
	left++

	a[pivp] = a[outerleft]

	sw := 0
	lwr := left
	for right-left >= unroll {
		for range unroll {
			x := a[left]
			h := b2i(cmp(x, piv) < 0)
			a[lwr] = x
			buf[sw] = x
			left++
			lwr += h
			sw += 1 - h
		}
	}
	for left <= right {
		x := a[left]
		h := b2i(cmp(x, piv) < 0)
		a[lwr] = x
		buf[sw] = x
		left++
		lwr += h
		sw += 1 - h
	}
	copy(a[lwr:], buf[:sw])
	lwr--
	a[outerleft] = a[lwr]
	a[lwr] = piv
	return lwr
}

func partitionFunc[T any](a []T, left, right int, buf []T, cmp func(a, b T) int) int {
	outerleft := left
	pivp := left + (right-left)/2
	piv := a[pivp]

	// Pivot: median of 7 medians of 5, as in the C version, but the 35 samples
	// are spread evenly over the range (sample 17 is pivp) instead of taken
	// in consecutive runs at both ends and the middle, which picked extreme
	// pivots on inputs with positional structure.
	j := pivotJitter(left, right)
	step := (right - left) / 36
	at := func(k int) int {
		return pivp + (k-17)*step + j
	}
	m5 := func(k int) {
		p0, p1, p2, p3, p4 := at(k), at(k+1), at(k+2), at(k+3), at(k+4)
		a[p0], a[p1], a[p2], a[p3], a[p4] = med5Func(cmp, a[p0], a[p1], a[p2], a[p3], a[p4])
	}
	m5(0)
	m5(5)
	m5(10)
	p15, p16, p18, p19 := at(15), at(16), at(18), at(19)
	a[p15], a[p16], piv, a[p18], a[p19] = med5Func(cmp, a[p15], a[p16], piv, a[p18], a[p19])
	m5(20)
	m5(25)
	m5(30)
	p2, p7, p12, p22, p27, p32 := at(2), at(7), at(12), at(22), at(27), at(32)
	a[p2], a[p7], a[p12], piv, a[p22], a[p27], a[p32] = med7Func(cmp, a[p2], a[p7], a[p12], piv, a[p22], a[p27], a[p32])

	left++
	a[pivp] = a[outerleft]

	rwr, sw := right, 0
	lwr := left

	for unroll < swsz-sw && left < right-unroll {
		avail := min(right-left, swsz-sw)
		endp := right - avail
		for right > endp+unroll {
			for range unroll {
				x := a[right]
				h := b2i(cmp(x, piv) < 0)
				a[rwr] = x
				buf[sw] = x
				right--
				rwr -= 1 - h
				sw += h
			}
		}
	}

	for right-left >= unroll && (rwr-right > unroll || left-lwr > unroll) {
		for rwr-right > unroll && right-left >= unroll {
			for range unroll {
				x := a[left]
				h := b2i(cmp(x, piv) < 0)
				a[lwr] = x
				a[rwr] = x
				left++
				lwr += h
				rwr -= 1 - h
			}
		}
		for left-lwr > unroll && right-left >= unroll {
			for range unroll {
				x := a[right]
				h := b2i(cmp(x, piv) < 0)
				a[rwr] = x
				a[lwr] = x
				right--
				rwr -= 1 - h
				lwr += h
			}
		}
	}
	for (lwr < left || rwr > right) && left <= right {
		for rwr > right && left <= right {
			x := a[left]
			left++
			if cmp(x, piv) < 0 {
				a[lwr] = x
				lwr++
			} else {
				a[rwr] = x
				rwr--
			}
		}
		for lwr < left && left <= right {
			x := a[right]
			right--
			if cmp(x, piv) < 0 {
				a[lwr] = x
				lwr++
			} else {
				a[rwr] = x
				rwr--
			}
		}
	}
	for left <= right {
		if cmp(a[left], piv) < 0 {
			left++
			lwr++
		} else {
			a[left], a[right] = a[right], a[left]
			right--
			rwr--
		}
	}
	copy(a[lwr:], buf[:sw])
	a[outerleft] = a[rwr]
	a[rwr] = piv
	return rwr
}

// smallSortFunc differs from the C version, which has no duplicate handling below
// SMALLPART and goes quadratic within a part when most elements equal the
// pivot. Like sortrFunc, a tiny left side collects the pivot's duplicates, splits
// that make no real progress count against an introsort-style limit, and
// recursion goes into the smaller side.
func smallSortFunc[T any](a []T, left, right int, buf []T, cmp func(a, b T) int) {
	limit := depthLimit(right - left + 1)
	for right-left > 17 {
		n := right - left
		// Part already sorted (pdqsort detects this too); costs about 16
		// compares otherwise.
		if runs, _ := scanOrderFunc(a[left:right+1], 1, cmp); runs == 1 {
			return
		}
		mid := partitionSmallFunc(a, left, right, buf, cmp)
		if mid-left < n/16 {
			smallSortFunc(a, left, mid-1, buf, cmp)
			piv := a[mid]
			mid++
			// collect duplicates
			for p := mid; p <= right; p++ {
				if cmp(piv, a[p]) >= 0 {
					a[mid], a[p] = a[p], a[mid]
					mid++
				}
			}
			if mid-left < n/16 {
				if limit--; limit < 0 {
					heapSortFunc(a[mid:right+1], cmp)
					return
				}
			}
			left = mid
			continue
		}
		if right-mid < n/16 {
			if limit--; limit < 0 {
				heapSortFunc(a[left:right+1], cmp)
				return
			}
		}
		if mid-left < right-mid {
			smallSortFunc(a, left, mid-1, buf, cmp)
			left = mid + 1
		} else {
			smallSortFunc(a, mid+1, right, buf, cmp)
			right = mid - 1
		}
	}
	sortingNetworkFunc(a[left:right+1], cmp)
}

// sortrFunc differs from the C version, which only detects a tiny left side and
// then gives one second chance before heapsorting the whole part. Here a tiny
// left side collects the pivot's duplicates (as in C), and any split that
// makes no real progress, on either side, counts against an introsort-style
// limit before falling back to heapsort.
func sortrFunc[T any](a []T, left, right int, buf []T, limit int, cmp func(a, b T) int) {
	for right-left > smallPart {
		n := right - left
		// Part already sorted (pdqsort detects this too); costs about 16
		// compares otherwise.
		if runs, _ := scanOrderFunc(a[left:right+1], 1, cmp); runs == 1 {
			return
		}
		mid := partitionFunc(a, left, right, buf, cmp)
		if mid-left < n/16 {
			if mid > left {
				sortrFunc(a, left, mid-1, buf, limit, cmp)
			}
			piv := a[mid]
			mid++
			// collect duplicates
			for p := mid; p <= right; p++ {
				if cmp(piv, a[p]) >= 0 {
					a[mid], a[p] = a[p], a[mid]
					mid++
				}
			}
			if mid-left < n/16 {
				if limit--; limit < 0 {
					heapSortFunc(a[mid:right+1], cmp)
					return
				}
			}
			left = mid
			continue
		}
		if right-mid < n/16 {
			if limit--; limit < 0 {
				heapSortFunc(a[left:right+1], cmp)
				return
			}
		}
		if mid-left < right-mid {
			sortrFunc(a, left, mid-1, buf, limit, cmp)
			left = mid + 1
		} else {
			sortrFunc(a, mid+1, right, buf, limit, cmp)
			right = mid - 1
		}
	}
	smallSortFunc(a, left, right, buf, cmp)
}

// scanOrderFunc counts the alternating non-decreasing / non-increasing runs in
// data, stopping once there are more than k. Each run is consumed by a tight
// loop whose branch is predictable inside a run (equal elements continue the
// run, so duplicates cost nothing), as in pdqsort's sortedness check; on
// unstructured data it gives up after about k comparisons. A drop between two
// ascending runs counts as a short descending run, so this over-counts by up
// to 2x; runLimit is calibrated for that. ascEnd is where the first
// non-decreasing run ends: runs == 1 means sorted, and runs == 2 &&
// ascEnd == 1 means non-increasing if k >= 2 (so the second run was scanned
// to the end).
func scanOrderFunc[T any](data []T, k int, cmp func(a, b T) int) (runs, ascEnd int) {
	n := len(data)
	i := 1
	for i < n && !(cmp(data[i], data[i-1]) < 0) {
		i++
	}
	runs, ascEnd = 1, i
	for i < n {
		if runs++; runs > k {
			break
		}
		if runs%2 == 0 {
			for i < n && !(cmp(data[i-1], data[i]) < 0) {
				i++
			}
		} else {
			for i < n && !(cmp(data[i], data[i-1]) < 0) {
				i++
			}
		}
	}
	return runs, ascEnd
}

// SortFunc sorts data in ascending order as determined by cmp, like
// slices.SortFunc. It is only fast if cmp compiles to branch-free code;
// cmp.Compare does not (it branches on the data). Prefer SortByKey or
// SortLess.
func SortFunc[T any](data []T, cmp func(a, b T) int) {
	n := len(data)
	if n < 2 {
		return
	}
	if n <= 18 {
		sortingNetworkFunc(data, cmp)
		return
	}
	// One pass detects sorted, reversed and few-runs input; on unstructured
	// data it stops after a few dozen comparisons.
	lim := max(runLimit(n), 2)
	switch runs, ascEnd := scanOrderFunc(data, lim, cmp); {
	case runs == 1:
		return
	case runs == 2 && ascEnd == 1: // complete scan, since lim >= 2
		// Non-increasing; reversing it sorts it.
		for i := 0; i < n/2; i++ {
			data[i], data[n-1-i] = data[n-1-i], data[i]
		}
		return
	case runs <= runLimit(n):
		// A few long runs: pdqsort is near-linear there, blqsort is not.
		slices.SortFunc(data, cmp)
		return
	}
	var buf [smallPart + 1]T
	sortrFunc(data, 0, n-1, buf[:], depthLimit(n), cmp)
}

func sort2Func[T any](v []T, cmp func(a, b T) int) {
	_ = v[1]
	a, b := v[0], v[1]

	v[0], v[1] = a, b
}
